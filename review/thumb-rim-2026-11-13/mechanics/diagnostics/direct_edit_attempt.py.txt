"""R19 replace upper/wrist-side thumb-mouth rounds on the native R18 frame."""

HERE=Path(__file__).resolve().parent;STUDY=HERE.parent;ROOT=STUDY.parents[1];BASE=ROOT/'review/organic-palm-2026-11-13'
s=importlib.util.spec_from_file_location('rim_r18',BASE/'mechanics/organic_palm.py');r18=importlib.util.module_from_spec(s);s.loader.exec_module(r18)
b,r17,prior,vec=r18.b,r18.r17,r18.prior,r18.vec
SOURCE=STUDY/'fingerworker_R19_thumb_rim.fs';BASE_SOURCE=BASE/'fingerworker_R18_organic_palm.fs';BASE_SHA='345b90327031276f8085e2c225c795946036d7faa4f2ac01b764101b9f4b3dd1';sha=r18.sha
RADIUS=2.0
rows=json.loads((HERE/'native-rim-faces.json').read_text())
POSITIVE_PICKS=[q['uv_midpoint'] for q in rows if q['index'] in [341,342,343,344,345,346]]
FACE_PICKS=[q['point_mm'] for q in json.loads((HERE/'native-face-picks.json').read_text())['faces']]
def load():
 inv=json.loads((BASE/'validation/native-step-inventory.json').read_text());assert sha(inv['native_step'])==inv['native_step_sha256'];ss,_,_=b.read_step(inv['native_step'])
 return {ref:ss[q['candidates'][0]['solid_index']] for ref,q in inv['native_part_correspondence'].items()},inv

def faces(shape):
 ex=TopExp_Explorer(shape,TopAbs_FACE);out=[]
 while ex.More():out.append(TopoDS.Face(ex.Current()));ex.Next()
 return out

def selected_faces(shape):
 fs=faces(shape);selected=[]
 for pt in FACE_PICKS:
  vertex=BRepBuilderAPI_MakeVertex(gp_Pnt(*pt)).Vertex();hits=[f for f in fs if b.gap(vertex,f)<1e-6]
  assert len(hits)==1,(pt,len(hits));selected+=hits
 assert len(selected)==12
 return selected

def defeature(shape):
 op=BRepAlgoAPI_Defeaturing();op.SetShape(shape)
 for f in selected_faces(shape):op.AddFaceToRemove(f)
 op.Build();assert op.IsDone(),'Defeaturing failed';out=op.Shape();assert BRepCheck_Analyzer(out).IsValid(),'Invalid restored mouth'
 return out

def mouth_edges(shape):
 out=[]
 for e in r17.edges(shape):
  a=BRepAdaptor_Curve(e);pts=[a.Value(a.FirstParameter()+(a.LastParameter()-a.FirstParameter())*t) for t in [.2,.5,.8]]
  if not all(abs(math.hypot(p.X()+24,abs(p.Y())-75)-14)<1e-4 for p in pts):continue
  bb=b.bounds(e)
  if bb[0,2]>27.99 or (bb[1,0]<-27 and bb[1,2]-bb[0,2]>5):out.append(e)
 return out

def build(old,radius=RADIUS):
 sharp=defeature(old['P01']);es=mouth_edges(sharp);assert es,'No upper/wrist rim edges';out=r17.rounded(sharp,es,radius)
 return out,sharp,[r17.middle(e) for e in es]

def expected(old=None):
 if old is None:old,_=load()
 return {'P01':build(old)[0]}

def patch_source():

 patch+='        const R19RimFaces=qUnion([\n'
 patch+=',\n'.join('            qClosestTo(qWithinRadius(qOwnedByBody(body(id + "base"),EntityType.FACE),'+vec(p)+',0.02*mm),'+vec(p)+')' for p in FACE_PICKS)
 patch+=']);\n        const R19RimFaceCount=size(evaluateQuery(context,R19RimFaces));\n'
 patch+='        println({"R19RimFaceCount":R19RimFaceCount});\n'
 patch+='        if (R19RimFaceCount != 12)\n            throw regenError("R19 thumb rim: expected12 distinct fillet faces within0.02mm of verified surface points");\n'

 return source.replace(anchor,patch+anchor).replace('Flat132 R18 - organic palm bolster and side thumb seats','Flat132 R19 - softened thumb rims and organic palm').replace('Fingerworker Flat132 R18 - organic palm bolster and integral side thumb seats.','Fingerworker Flat132 R19 - generous upper thumb rims and organic palm.').replace('P01 PRINT STUDY - Flat132 organic palm bolster, side thumb seats, inboard straps','P01 PRINT STUDY - Flat132 rounded thumb rims, organic palm, inboard straps')

def run():
 old,inv=load();SOURCE.write_text(patch_source());sharp=defeature(old['P01']);prior.export(sharp,HERE/'P01_restored_upper_rim');es=mouth_edges(sharp)
 (HERE/'selection.json').write_text(json.dumps(dict(face_points_mm=FACE_PICKS,sharp_edge_points_mm=[r17.middle(e) for e in es],edges=len(es)),indent=2)+'\n')
 for rad in [2.0,1.5]:
  try:
   frame=r17.rounded(sharp,es,rad);mesh=r17.export_mesh(frame,HERE/f'P01_rim_R{str(rad).replace(".","p")}')
   result=dict(status='VALID_LOCAL_RIM_CANDIDATE',radius_mm=rad,mesh=mesh);(HERE/'trial-result.json').write_text(json.dumps(result,indent=2)+'\n');break
  except Exception as e:(HERE/f'trial-R{rad}.txt').write_text(str(e)+'\n')
 else:raise RuntimeError('Neither R2 nor R1.5 blend succeeded')
